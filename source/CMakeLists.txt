# CMake build for AssaultCube on Linux and macOS (an alternative to source/src/Makefile; both build the same programs).
#
#   cmake -S source -B build                  # Release by default
#   cmake --build build -j
#
# Produces build/ac_client, build/ac_server and build/ac_master; run the client from the repository root, e.g.
#   build/ac_client --home="$HOME/.assaultcube/v1.4" --init
# For a distributable, self-contained macOS app use source/macos/make-app.sh instead.
#
# Options: -DAC_CLIENT=OFF / -DAC_SERVER=OFF / -DAC_MASTER=OFF, -DAC_LTO=ON, -DAC_MARCH=x86-64-v3
cmake_minimum_required(VERSION 3.16)
project(AssaultCube LANGUAGES C CXX)

option(AC_CLIENT "Build the game client" ON)
option(AC_SERVER "Build the standalone server" ON)
option(AC_MASTER "Build the master server" ON)
option(AC_LTO "Link-time optimisation" OFF)
set(AC_MARCH "" CACHE STRING "CPU baseline passed as -march (e.g. x86-64-v2), empty for the compiler default")

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS ON)
if(AC_LTO)
  set(CMAKE_INTERPROCEDURAL_OPTIMIZATION ON)
endif()

set(SRC ${CMAKE_CURRENT_SOURCE_DIR}/src)
set(ENET ${CMAKE_CURRENT_SOURCE_DIR}/enet)

# same flags as the Makefile: signed char everywhere (it is unsigned on Linux/ARM), no -ffast-math (physics is
# predicted on every client, so all builds must round alike)
add_compile_options(-fsigned-char -Wall $<$<CONFIG:Release>:-O3> $<$<CONFIG:Release>:-fomit-frame-pointer>)
if(AC_MARCH)
  add_compile_options(-march=${AC_MARCH})
endif()

# --- ENet (bundled, built as a static library; feature checks replace its autotools configure)
include(CheckSymbolExists)
include(CheckStructHasMember)
include(CheckTypeSize)
check_symbol_exists(poll "poll.h" HAS_POLL)
check_symbol_exists(fcntl "fcntl.h" HAS_FCNTL)
check_symbol_exists(inet_pton "arpa/inet.h" HAS_INET_PTON)
check_symbol_exists(inet_ntop "arpa/inet.h" HAS_INET_NTOP)
check_symbol_exists(gethostbyname_r "netdb.h" HAS_GETHOSTBYNAME_R)
check_symbol_exists(gethostbyaddr_r "netdb.h" HAS_GETHOSTBYADDR_R)
check_struct_has_member("struct msghdr" msg_flags "sys/socket.h" HAS_MSGHDR_FLAGS)
set(CMAKE_EXTRA_INCLUDE_FILES "sys/socket.h")
check_type_size(socklen_t SOCKLEN_T)
unset(CMAKE_EXTRA_INCLUDE_FILES)
add_library(enet STATIC
  ${ENET}/callbacks.c ${ENET}/compress.c ${ENET}/host.c ${ENET}/list.c
  ${ENET}/packet.c ${ENET}/peer.c ${ENET}/protocol.c ${ENET}/unix.c)
target_include_directories(enet PUBLIC ${ENET}/include)
foreach(def HAS_POLL HAS_FCNTL HAS_INET_PTON HAS_INET_NTOP HAS_GETHOSTBYNAME_R HAS_GETHOSTBYADDR_R HAS_MSGHDR_FLAGS)
  if(${def})
    target_compile_definitions(enet PRIVATE ${def})
  endif()
endforeach()
if(HAVE_SOCKLEN_T)
  target_compile_definitions(enet PRIVATE HAS_SOCKLEN_T)
endif()

find_package(ZLIB REQUIRED)
find_package(Threads REQUIRED)
find_package(PkgConfig QUIET)

# A dependency as an imported target: the library's own CMake package if it has one, else pkg-config, else a plain
# search for its libraries and header (pkg-config isn't always installed, and libvorbis ships no CMake package).
# ac_dependency(<target> <cmake package> <cmake target> <pkg-config module> <header> <header subdir> <libraries...>)
function(ac_dependency target package packagetarget pcmodule header headersuffix)
  string(MAKE_C_IDENTIFIER "${target}" id) # variable-name-safe version of the target name (which contains "::")
  if(package)
    find_package(${package} CONFIG QUIET)
    if(TARGET ${packagetarget})
      add_library(${target} INTERFACE IMPORTED)
      target_link_libraries(${target} INTERFACE ${packagetarget})
      return()
    endif()
  endif()
  if(PKG_CONFIG_FOUND)
    pkg_check_modules(PC_${id} QUIET IMPORTED_TARGET ${pcmodule})
    if(PC_${id}_FOUND)
      add_library(${target} INTERFACE IMPORTED)
      target_link_libraries(${target} INTERFACE PkgConfig::PC_${id})
      return()
    endif()
  endif()
  find_path(${id}_INCLUDE_DIR NAMES ${header} PATH_SUFFIXES ${headersuffix})
  set(libs)
  foreach(lib ${ARGN})
    find_library(${id}_${lib}_LIBRARY NAMES ${lib})
    if(NOT ${id}_${lib}_LIBRARY)
      message(FATAL_ERROR "AssaultCube needs ${lib} (library for ${pcmodule}): not found")
    endif()
    list(APPEND libs ${${id}_${lib}_LIBRARY})
  endforeach()
  if(NOT ${id}_INCLUDE_DIR)
    message(FATAL_ERROR "AssaultCube needs ${header} (${pcmodule}): not found")
  endif()
  add_library(${target} INTERFACE IMPORTED)
  target_link_libraries(${target} INTERFACE ${libs})
  target_include_directories(${target} INTERFACE ${${id}_INCLUDE_DIR})
endfunction()

# SDL3: the client always, the server on macOS (it uses SDL threads there: macOS has no unnamed POSIX semaphores)
if(AC_CLIENT OR APPLE)
  ac_dependency(ac::sdl3 SDL3 SDL3::SDL3 sdl3 SDL3/SDL.h "" SDL3)
endif()

set(SERVER_SOURCES ${SRC}/crypto.cpp ${SRC}/log.cpp ${SRC}/protocol.cpp ${SRC}/serverms.cpp ${SRC}/server.cpp
  ${SRC}/stream.cpp ${SRC}/tools.cpp ${SRC}/wizard.cpp)

if(AC_CLIENT)
  find_package(OpenGL REQUIRED)
  ac_dependency(ac::vorbisfile "" "" vorbisfile vorbis/vorbisfile.h "" vorbisfile vorbis ogg)
  find_package(OpenAL REQUIRED)
  file(GLOB CLIENT_SOURCES ${SRC}/*.cpp ${SRC}/bot/*.cpp)
  list(REMOVE_ITEM CLIENT_SOURCES ${SRC}/master.cpp ${SRC}/pch.cpp)
  add_executable(ac_client ${CLIENT_SOURCES})
  target_include_directories(ac_client PRIVATE ${SRC} ${SRC}/bot)
  # source/include holds Windows copies of SDL etc.; only its GL/glext.h is wanted, so it's searched last (as in the Makefile)
  target_compile_options(ac_client PRIVATE -idirafter ${CMAKE_CURRENT_SOURCE_DIR}/include)
  target_include_directories(ac_client PRIVATE ${OPENAL_INCLUDE_DIR})
  target_link_libraries(ac_client PRIVATE enet ac::sdl3 ac::vorbisfile ZLIB::ZLIB
    OpenGL::GL ${OPENAL_LIBRARY} Threads::Threads)
  if(UNIX AND NOT APPLE)
    find_package(X11 REQUIRED)
    target_link_libraries(ac_client PRIVATE X11::X11)
  endif()
  target_precompile_headers(ac_client PRIVATE ${SRC}/cube.h)
  set_source_files_properties(${SRC}/stbimage.cpp PROPERTIES SKIP_PRECOMPILE_HEADERS ON) # stb_image collides with cube.h's macros
endif()

if(AC_SERVER)
  add_executable(ac_server ${SERVER_SOURCES})
  target_compile_definitions(ac_server PRIVATE STANDALONE)
  target_include_directories(ac_server PRIVATE ${SRC} ${SRC}/bot)
  target_link_libraries(ac_server PRIVATE enet ZLIB::ZLIB Threads::Threads)
  if(APPLE)
    target_link_libraries(ac_server PRIVATE ac::sdl3)
  elseif(NOT CMAKE_SYSTEM_NAME STREQUAL "OpenBSD")
    target_link_libraries(ac_server PRIVATE rt)
  endif()
endif()

  add_executable(ac_master ${SRC}/stream.cpp ${SRC}/command.cpp ${SRC}/tools.cpp ${SRC}/protocol.cpp ${SRC}/master.cpp)
  target_compile_definitions(ac_master PRIVATE STANDALONE)
  target_include_directories(ac_master PRIVATE ${SRC} ${SRC}/bot)
  target_link_libraries(ac_master PRIVATE enet ZLIB::ZLIB Threads::Threads)
  if(APPLE)
    target_link_libraries(ac_master PRIVATE ac::sdl3)
  elseif(NOT CMAKE_SYSTEM_NAME STREQUAL "OpenBSD")
    target_link_libraries(ac_master PRIVATE rt)
  endif()
endif()
